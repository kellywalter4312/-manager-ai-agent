# Runtime dependencies: none for the default configuration.
# The system runs entirely on the standard library with MANAGER_PROVIDER=null.

# Optional - only needed for MANAGER_PROVIDER=anthropic.
# It is imported lazily, so the project runs without it installed.
anthropic>=0.69

# Optional - only needed for MANAGER_PROVIDER=openai or MANAGER_PROVIDER=xai
# (xAI's API is OpenAI-compatible and is reached through this same SDK,
# pointed at xAI's own base URL - see src/providers/xai_provider.py).
# Also imported lazily; the project runs without it installed.
openai>=1.0

# Optional - only needed for MANAGER_MAILBOX=gmail (read-only access).
# Also imported lazily; the full test suite passes without these.
google-api-python-client>=2.100
google-auth-oauthlib>=1.2

# Optional - only needed for MANAGER_SPEECH=kokoro or MANAGER_MEDIA=local
# (the latter reuses the identical engine for narration - see
# src/providers/media/narration.py). A genuine pre-existing gap in this
# file, found while documenting the entry below: this was already a real
# runtime dependency and was never listed here.
kokoro-onnx>=0.4

# Optional - only needed for MANAGER_MEDIA=local (real narration/image/video
# production - see src/providers/media/). Pillow renders local title-card
# images; imageio-ffmpeg bundles a real, statically-linked FFmpeg binary so
# video composition needs no system-wide install. All imported lazily; the
# project runs without them installed.
Pillow>=10.0
imageio-ffmpeg>=0.5

# Optional - only needed to read .pdf files placed in the user's own
# knowledge_dir (see src/providers/document_knowledge_provider.py). Real
# text extraction, page by page - never OCR. Imported lazily; the
# project, and every other document format, runs without it installed.
pypdf>=4.0

# Optional - only needed to read .xlsx files placed in the user's own
# knowledge_dir. Real cell data, row by row, per sheet. Imported lazily;
# .csv needs no dependency at all (the standard library's own csv module).
openpyxl>=3.1

# Optional - only needed to read .pptx files placed in the user's own
# knowledge_dir. Real slide text, never OCR on slide images. Imported
# lazily.
python-pptx>=0.6

# Optional - only needed for `scripts/serve_ui.py --https` (see
# scripts/dev_tls.py). Generates the private, locally-signed TLS
# certificate that lets the user's own other devices - a phone on the
# same private network - use microphone-dependent capabilities, which
# browsers otherwise refuse outside a secure context. Imported lazily;
# the project, and plain-HTTP --host, run without it installed. Already
# present transitively via google-auth (MANAGER_MAILBOX=gmail) in many
# installs, but not guaranteed, so it is listed here explicitly rather
# than relied on by accident.
cryptography>=41.0

# Optional - only needed for literal website interaction
# (MANAGER_CONNECTION_PROVIDER=<a declared site> - see
# src/providers/browser/cdp_browser.py). Speaks the Chrome DevTools
# Protocol to an already-installed, real Chromium-based browser (Edge or
# Chrome) - no browser binary is bundled or downloaded, and no browser-
# automation framework (Playwright/Selenium) is depended on. Imported
# lazily; the project, and every other connection kind, run without it
# installed.
websockets>=12.0

# Development dependencies:
pytest>=8.0

# Optional - local models on this machine (MANAGER_PROVIDER=local,
# MANAGER_LOCAL_ASSIST=local, offline translation, local multilingual speech
# recognition for live translation - MANAGER_VIDEO_UNDERSTANDING=local_whisper).
# Model files live under data/models (not tracked). llama-cpp-python needs the
# prebuilt CPU wheel index on Windows:
#   pip install llama-cpp-python --prefer-binary --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cpu
# argostranslate's wheel ships a stray top-level "tests" package into
# site-packages that shadows this repository's tests - delete
# .venv/Lib/site-packages/tests after installing it.
llama-cpp-python>=0.3
faster-whisper>=1.0
argostranslate>=1.9
